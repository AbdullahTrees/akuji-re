# Mutations for the event interpreter. Run with
#
#     SELFTEST=--selftest-runner tools/mutate.sh tools/mutations/runner.txt
#
# Every one of these is a reading that was actually considered and rejected
# while translating the four functions, so each is a defect a careless
# translation could plausibly have shipped rather than an arbitrary edit.
===
the alternative scan runs forwards
src/EventRunner.pas
---
    for I := List.Count - 1 downto 0 do
--->
    for I := 0 to List.Count - 1 do
===
the alternative scan takes the first match rather than the last
src/EventRunner.pas
---
      if (Flag >= 0) and (Flag < PROGRESS_LENGTH)
         and (P.Progress[Flag] = 1) then
      begin
        Steps[StepIndex] := List[I];
        Exit;
      end;
      Steps[StepIndex] := '';
--->
      if (Flag >= 0) and (Flag < PROGRESS_LENGTH)
         and (P.Progress[Flag] = 1) then
        Steps[StepIndex] := List[I]
      else
        Steps[StepIndex] := '';
